    <script>
        document.addEventListener('DOMContentLoaded', () => {
            // Inicializa os ícones
            lucide.createIcons();

            // Elementos do DOM
            const form = document.getElementById('expense-form');
            const isParceladoCheckbox = document.getElementById('is-parcelado');
            const parcelasInfoDiv = document.getElementById('parcelas-info');
            const expenseList = document.getElementById('expense-list');
            const emptyStateRow = document.getElementById('empty-state');

            // Carrega os gastos do Local Storage ou inicializa um array vazio
            let expenses = JSON.parse(localStorage.getItem('expenses')) || [];

            // Função para formatar valores como moeda (Real Brasileiro)
            const formatCurrency = (value) => {
                return new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(value);
            };

            // Função para renderizar os gastos na tabela
            const renderExpenses = () => {
                expenseList.innerHTML = ''; // Limpa a lista
                if (expenses.length === 0) {
                    emptyStateRow.classList.remove('hidden');
                } else {
                    emptyStateRow.classList.add('hidden');
                    expenses.sort((a, b) => new Date(b.data) - new Date(a.data)); // Ordena por data mais recente
                    expenses.forEach(expense => {
                        const valorParcela = expense.valorTotal / (expense.totalParcelas || 1);
                        
                        const statusClass = expense.status === 'Pago' ? 'bg-green-100 text-green-800' : 'bg-yellow-100 text-yellow-800';
                        const statusText = expense.status;
                        
                        const newRow = document.createElement('tr');
                        newRow.className = 'bg-white hover:bg-slate-50 transition-colors';
                        newRow.innerHTML = `
                            <td class="px-6 py-4 font-medium text-slate-900">${expense.descricao}</td>
                            <td class="px-6 py-4">
                                <div class="font-semibold">${formatCurrency(valorParcela)}</div>
                                <div class="text-xs text-slate-500">${formatCurrency(expense.valorTotal)} total</div>
                            </td>
                            <td class="px-6 py-4">${expense.parcelasPagas} / ${expense.totalParcelas}</td>
                            <td class="px-6 py-4">
                                <button onclick="toggleStatus('${expense.id}')" title="Clique para alterar" class="w-24 text-center cursor-pointer px-2 py-1 text-xs font-medium rounded-full ${statusClass} transition-transform transform hover:scale-105">${statusText}</button>
                            </td>
                            <td class="px-6 py-4">${expense.quemPagou}</td>
                            <td class="px-6 py-4">${new Date(expense.data + 'T03:00:00Z').toLocaleDateString('pt-BR')}</td>
                            <td class="px-6 py-4 text-center">
                                <div class="flex items-center justify-center gap-2">
                                    ${expense.isParcelado && expense.parcelasPagas > 0 ? `<button title="Remover pagamento de parcela" onclick="unpayInstallment('${expense.id}')" class="p-1 text-orange-500 hover:text-orange-700 transition-colors"><i data-lucide="minus-circle" class="w-5 h-5"></i></button>` : `<div class="w-7 h-7"></div>`}
                                    ${expense.status === 'Pendente' ? `<button title="Pagar uma parcela" onclick="payInstallment('${expense.id}')" class="p-1 text-green-600 hover:text-green-800 transition-colors"><i data-lucide="plus-circle" class="w-5 h-5"></i></button>` : `<div class="w-7 h-7"></div>`}
                                    <button title="Excluir gasto" onclick="deleteExpense('${expense.id}')" class="p-1 text-red-500 hover:text-red-700 transition-colors"><i data-lucide="trash-2" class="w-5 h-5"></i></button>
                                </div>
                            </td>
                        `;
                        expenseList.appendChild(newRow);
                    });
                }
                updateSummary();
                lucide.createIcons(); // Recria os ícones após renderizar
            };

            // Função para atualizar o resumo financeiro
            const updateSummary = () => {
                const hoje = new Date();
                const mesAtual = hoje.getMonth();
                const anoAtual = hoje.getFullYear();

                // 1. Total de dívidas futuras (parcelas restantes)
                const totalAPagar = expenses.reduce((acc, exp) => {
                    if (exp.status === 'Pendente' && exp.isParcelado) {
                        const valorParcela = exp.valorTotal / exp.totalParcelas;
                        const parcelasRestantes = exp.totalParcelas - exp.parcelasPagas;
                        return acc + (valorParcela * parcelasRestantes);
                    }
                    return acc;
                }, 0);

                // 2. Desembolso real no mês (fluxo de caixa)
                let desembolsoMes = 0;
                expenses.forEach(exp => {
                    const dataGasto = new Date(exp.data + 'T03:00:00Z');
                    if (exp.isParcelado) {
                        const purchaseMonth = dataGasto.getMonth();
                        const purchaseYear = dataGasto.getFullYear();
                        const monthsSincePurchase = (anoAtual - purchaseYear) * 12 + (mesAtual - purchaseMonth);
                        if (monthsSincePurchase >= 0 && monthsSincePurchase < exp.totalParcelas && exp.status === 'Pendente') {
                            const valorParcela = exp.valorTotal / exp.totalParcelas;
                            desembolsoMes += valorParcela;
                        }
                    } else { // Compra à vista
                        if (dataGasto.getMonth() === mesAtual && dataGasto.getFullYear() === anoAtual && exp.status === 'Pendente') {
                             desembolsoMes += exp.valorTotal;
                        }
                    }
                });
                
                // 3. Total de compras feitas no mês (valor total dos itens)
                const totalComprasMes = expenses.reduce((acc, exp) => {
                    const dataGasto = new Date(exp.data + 'T03:00:00Z');
                    if (dataGasto.getMonth() === mesAtual && dataGasto.getFullYear() === anoAtual) {
                        return acc + exp.valorTotal;
                    }
                    return acc;
                }, 0);

                document.getElementById('total-a-pagar').textContent = formatCurrency(totalAPagar);
                document.getElementById('desembolso-mes').textContent = formatCurrency(desembolsoMes);
                document.getElementById('total-compras-mes').textContent = formatCurrency(totalComprasMes);
            };

            // Função para salvar no Local Storage
            const saveExpenses = () => {
                localStorage.setItem('expenses', JSON.stringify(expenses));
            };

            // Event listener para o checkbox de parcelamento
            isParceladoCheckbox.addEventListener('change', () => {
                parcelasInfoDiv.classList.toggle('show-details', isParceladoCheckbox.checked);
                parcelasInfoDiv.classList.toggle('hidden-details', !isParceladoCheckbox.checked);
            });

            // Event listener para o envio do formulário
            form.addEventListener('submit', (e) => {
                e.preventDefault();

                const isParcelado = isParceladoCheckbox.checked;
                const totalParcelasInput = document.getElementById('total-parcelas');
                const parcelasPagasInput = document.getElementById('parcelas-pagas');
                
                if (isParcelado && (!totalParcelasInput.value || parseInt(totalParcelasInput.value) <= 0)) {
                    alert('Por favor, insira um número válido de parcelas.');
                    return;
                }
                
                const newExpense = {
                    id: Date.now().toString(),
                    descricao: document.getElementById('descricao').value,
                    valorTotal: parseFloat(document.getElementById('valor').value),
                    data: document.getElementById('data').value,
                    quemPagou: document.getElementById('quem-pagou').value,
                    isParcelado: isParcelado,
                    totalParcelas: isParcelado ? parseInt(totalParcelasInput.value) || 1 : 1,
                    parcelasPagas: isParcelado ? parseInt(parcelasPagasInput.value) || 0 : 0,
                };
                newExpense.status = newExpense.parcelasPagas >= newExpense.totalParcelas ? 'Pago' : 'Pendente';


                expenses.push(newExpense);
                saveExpenses();
                renderExpenses();
                form.reset();
                isParceladoCheckbox.checked = false;
                parcelasInfoDiv.classList.remove('show-details');
                parcelasInfoDiv.classList.add('hidden-details');
            });
            
            // Funções globais para os botões de ação
            window.toggleStatus = (id) => {
                const expense = expenses.find(exp => exp.id === id);
                if (expense) {
                    if (expense.status === 'Pendente') {
                        expense.status = 'Pago';
                        expense.parcelasPagas = expense.totalParcelas;
                    } else { // status was 'Pago'
                        expense.status = 'Pendente';
                        if (expense.totalParcelas > 0) {
                            expense.parcelasPagas = expense.totalParcelas - 1;
                        }
                    }
                    saveExpenses();
                    renderExpenses();
                }
            };

            window.unpayInstallment = (id) => {
                const expense = expenses.find(exp => exp.id === id);
                if (expense && expense.isParcelado && expense.parcelasPagas > 0) {
                    expense.parcelasPagas--;
                    if (expense.parcelasPagas < expense.totalParcelas) {
                        expense.status = 'Pendente';
                    }
                    saveExpenses();
                    renderExpenses();
                }
            };

            window.payInstallment = (id) => {
                const expense = expenses.find(exp => exp.id === id);
                if (expense && expense.parcelasPagas < expense.totalParcelas) {
                    expense.parcelasPagas++;
                    if (expense.parcelasPagas >= expense.totalParcelas) {
                        expense.status = 'Pago';
                    }
                    saveExpenses();
                    renderExpenses();
                }
            };

            window.deleteExpense = (id) => {
                // A exclusão agora é imediata, sem confirmação.
                expenses = expenses.filter(exp => exp.id !== id);
                saveExpenses();
                renderExpenses();
            };



            // Renderização inicial
            renderExpenses();
        });
    </script>
